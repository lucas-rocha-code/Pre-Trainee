//Exercicio 2 "Aluno aprovado ou reprovado?"

algoritmo "aprovação"

var
    nota1, nota2, media: Real
inicio
    Escreva("Primeira nota: ")
    Leia(nota1)
    Escreva("Segunda nota: ")
    Leia(nota2)
    media <- (nota1+nota2)/2
    Se media >= 7 entao
        Escreva("Aluno aprovado! a média foi: ", media)
    Senao
        Escreva("Aluno reprovado, a média foi: ", media)
    fimse
fimalgoritmo
